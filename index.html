/** 第一次使用：建好三個分頁、裝好勾選框，並且馬上抓第一次 */
function setup() {
  const ss = SpreadsheetApp.getActive();
  prepareSettingsSheet_(ss);
  prepareDataSheet_(ss);
  prepareBoardSheet_(ss);
  installEditTrigger_(ss);
  ss.getSheetByName(SHEETS.settings).activate();
  runFetch_();
  ss.toast('設定完成！到「設定」分頁看狀態列', '完成', 5);
}

/** 「設定」分頁有人改東西時，Google 會自動呼叫這裡（不用手動執行） */
function onSettingsEdit(e) {
  const range = e && e.range;
  if (!range || range.getSheet().getName() !== SHEETS.settings) return;
  const cell = range.getA1Notation();
  if (cell === CELLS.runNow && range.getValue() === true) {
    range.setValue(false); // 勾完自動取消，下次可以再勾
    runFetch_();
  } else if (cell === CELLS.daily) {
    setDailyTrigger_(range.getValue() === true);
  }
}

/** 把一類的相關字塞進來源網址 */
function buildFeedUrl_(words, days) {
  if (來源網址.indexOf('{關鍵字}') < 0) return 來源網址; // 固定網址的來源，整包收下
  const terms = words.map((w) => (w.indexOf(' ') >= 0 ? '"' + w + '"' : w));
  const query = '(' + terms.join(' OR ') + ') when:' + days + 'd';
  return 來源網址.replace('{關鍵字}', encodeURIComponent(query));
}

/** 把來源回傳的內容（RSS），照②的對照表拆成一筆一筆 */
function parseFeed_(xml) {
  const blocks = xml.match(/<item>[\s\S]*?<\/item>/g) || [];
  const items = [];
  blocks.forEach((block) => {
    const id = pickTag_(block, 要抓的項目.身分證);
    const source = 要抓的項目.媒體 ? pickTag_(block, 要抓的項目.媒體) : '';
    let title = pickTag_(block, 要抓的項目.標題);
    const suffix = ' - ' + source; // 有些來源的標題後面會接「 - 媒體名」，拿掉
    if (source && title.endsWith(suffix)) title = title.slice(0, -suffix.length);
    if (id && title) {
      items.push({
        id: id,
        title: title,
        source: source,
        published: new Date(pickTag_(block, 要抓的項目.時間)),
      });
    }
  });
  return items;
}

/** 每一類各抓一次，一次同時送出 */
function fetchAll_(categories, days) {
  const requests = categories.map((c) => ({ url: buildFeedUrl_(c.words, days), muteHttpExceptions: true }));
  const responses = UrlFetchApp.fetchAll(requests);
  return categories.map((c, i) => {
    const res = responses[i];
    if (res.getResponseCode() !== 200) return { category: c.name, items: [], failed: true };
    return { category: c.name, items: parseFeed_(res.getContentText('UTF-8')), failed: false };
  });
}

// ============================================================
// 三、整理資料：② 認身分證 → ③ 分群組 → ④ 貼標註
//     這一段只處理「一筆一筆的資料」，不管資料是從哪裡抓來的
// ============================================================

/** 把各類抓回來的合併：同一個網址只留一筆，並記下它是從哪幾類搜到的 */
function mergeBatch_(results, days, now) {
  const cutoff = now.getTime() - days * 24 * 3600 * 1000;
  const byId = {};
  const order = [];
  results.forEach((r) => {
    r.items.forEach((item) => {
      if (item.published.getTime() < cutoff) return; // 超過時間區間的不要
      if (!byId[item.id]) {
        byId[item.id] = Object.assign({ fromCategories: [] }, item);
        order.push(item.id);
      }
      if (byId[item.id].fromCategories.indexOf(r.category) < 0) byId[item.id].fromCategories.push(r.category);
    });
  });
  const time = (item) => item.published.getTime() || 0;
  return order.map((id) => byId[id]).sort((a, b) => time(b) - time(a)); // 新的排前面
}

/** ② 認身分證：網址已經在資料庫裡的，不加列、也不做任何動作，只算數量 */
function splitNewAndSeen_(items, existingIds) {
  const fresh = [];
  let seenCount = 0;
  items.forEach((item) => {
    if (existingIds.has(item.id)) seenCount++;
    else fresh.push(item);
  });
  return { fresh: fresh, seenCount: seenCount };
}

/**
 * ③ 分群組：標題很像的（同一件事、不同媒體），給同一個群組編號
 *   fresh：這次新抓到的；recent：資料庫裡最新的幾筆 { title, group }
 *   回傳 updates：recent 裡原本沒有群組、這次被配成一組的，要補寫回資料分頁
 */
function assignGroups_(fresh, recent, nextNo) {
  const pool = recent.map((r, i) => ({ sig: titleSignature_(r.title), ref: r, recentIndex: i }));
  const updates = [];
  fresh.forEach((item) => {
    const sig = titleSignature_(item.title);
    let best = null;
    let bestScore = 0;
    pool.forEach((p) => {
      const score = similarity_(sig, p.sig);
      if (score >= SAME_STORY_THRESHOLD && score > bestScore) {
        best = p;
        bestScore = score;
      }
    });
    item.group = '';
    if (best) {
      if (!best.ref.group) {
        best.ref.group = '第' + nextNo + '組';
        nextNo++;
        if (best.recentIndex !== undefined) updates.push({ index: best.recentIndex, group: best.ref.group });
      }
      item.group = best.ref.group;
    }
    pool.push({ sig: sig, ref: item });
  });
  return { updates: updates, nextNo: nextNo };
}

/** 把標題拆成「每兩個字一組」，拿來比對像不像（標點、空白先拿掉） */
function titleSignature_(title) {
  const s = String(title).toLowerCase().replace(/[^\p{L}\p{N}]+/gu, '');
  const grams = new Set();
  for (let i = 0; i < s.length - 1; i++) grams.add(s.slice(i, i + 2));
  return grams;
}

/** 兩個標題的共同字比例：0 = 完全不像，1 = 一模一樣 */
function similarity_(a, b) {
  if (!a.size || !b.size) return 0;
  const small = a.size <= b.size ? a : b;
  const large = small === a ? b : a;
  let shared = 0;
  small.forEach((g) => {
    if (large.has(g)) shared++;
  });
  return shared / (a.size + b.size - shared);
}

/** 找出目前用到第幾組，新的一組從下一號開始 */
function nextGroupNo_(groups) {
  let max = 0;
  groups.forEach((g) => {
    const m = String(g).match(/\d+/);
    if (m) max = Math.max(max, Number(m[0]));
  });
  return max + 1;
}

/** ④ 貼標註：從哪一類搜到就貼那一類；標題又提到別類的相關字，也一起貼 */
function tagsFor_(item, categories) {
  return categories
    .filter((c) => item.fromCategories.indexOf(c.name) >= 0 ||
      c.words.some((w) => titleHasWord_(item.title, w)))
    .map((c) => '【' + c.name + '】')
    .join('');
}

/** 標題裡有沒有這個相關字：中文直接找；英文要是完整的一個字，避免 TAIWAN 裡的 AI 被當成 AI */
function titleHasWord_(title, word) {
  const t = String(title).toLowerCase();
  const w = String(word).toLowerCase();
  if (!/^[a-z0-9 ]+$/.test(w)) return t.indexOf(w) >= 0;
  return new RegExp('(^|[^a-z0-9])' + w + '([^a-z0-9]|$)').test(t);
}

/** 關聯看板要的數字：各類幾則、這次新抓到幾則、哪些事被多家媒體報導 */
function summarize_(rows, freshIds, categories) {
  const counts = categories.map((c) => ({ name: c.name, total: 0, fresh: 0 }));
  const groups = {};
  rows.forEach((r) => {
    const tags = String(r[COL.tags]);
    const isFresh = freshIds.has(r[COL.id]);
    counts.forEach((c) => {
      if (tags.indexOf('【' + c.name + '】') < 0) return;
      c.total++;
      if (isFresh) c.fresh++;
    });
    const g = r[COL.group];
    if (!g) return;
    const entry = groups[g] || (groups[g] = { group: g, count: 0, title: r[COL.title], sources: [], tags: [] });
    entry.count++;
    if (entry.sources.indexOf(r[COL.source]) < 0) entry.sources.push(r[COL.source]);
    (tags.match(/【[^】]+】/g) || []).forEach((t) => {
      if (entry.tags.indexOf(t) < 0) entry.tags.push(t);
    });
  });
  const topGroups = Object.keys(groups)
    .map((k) => groups[k])
    .filter((g) => g.count >= 2)
    .sort((a, b) => b.count - a.count)
    .slice(0, 30)
    .map((g) => ({
      group: g.group,
      count: g.count,
      title: g.title,
      tags: g.tags.join(''),
      sources: g.sources.slice(0, 5).join('、') + (g.sources.length > 5 ? ' 等 ' + g.sources.length + ' 家' : ''),
    }));
  return { counts: counts, topGroups: topGroups };
}

/** 把相關字拆開：頓號、逗號、分號、斜線、換行都可以；重複的只留一個 */
function parseWords_(text) {
  const seen = {};
  return String(text == null ? '' : text)
    .split(/[、,，;；\/\n]+/)
    .map((w) => w.trim())
    .filter((w) => {
      if (!w || seen[w]) return false;
      seen[w] = true;
      return true;
    });
}

// ============================================================
// 四、主流程：① 抓取 → ② 認身分證 → ③ 分群組 → ④ 貼標註 → ⑤ 寫入並亮燈
// ============================================================

function runFetch_() {
  const ss = SpreadsheetApp.getActive();
  const lock = LockService.getDocumentLock();
  if (!lock.tryLock(1000)) {
    setStatus_(ss, '上一次還在抓，請等它跑完再勾');
    return;
  }
  try {
    const settings = readSettings_(ss);
    if (!settings.categories.length) {
      setStatus_(ss, '分類表是空的：請在下面的分類表至少填一列');
      return;
    }
    setStatus_(ss, '抓取中⋯請稍候（約 10～30 秒）');
    SpreadsheetApp.flush();

    const now = new Date();
    const results = fetchAll_(settings.categories, settings.days); // ① 抓取
    const batch = mergeBatch_(results, settings.days, now);

    const dataSheet = ss.getSheetByName(SHEETS.data);
    const existing = readDataRows_(dataSheet);
    const split = splitNewAndSeen_(batch, new Set(existing.map((r) => r[COL.id]))); // ② 認身分證

    const recent = existing.slice(0, COMPARE_WINDOW).map((r) => ({ title: r[COL.title], group: r[COL.group] }));
    const grouping = assignGroups_(split.fresh, recent, nextGroupNo_(existing.map((r) => r[COL.group]))); // ③ 分群組
    split.fresh.forEach((item) => {
      item.tags = tagsFor_(item, settings.categories); // ④ 貼標註
    });

    const newRows = split.fresh.map((item) => toRow_(item, now));
    writeResults_(dataSheet, newRows, grouping.updates.length ? recent : null); // ⑤ 寫入並亮燈

    recent.forEach((r, i) => {
      existing[i][COL.group] = r.group;
    });
    const freshIds = new Set(newRows.map((r) => r[COL.id]));
    writeBoard_(ss, summarize_(newRows.concat(existing), freshIds, settings.categories));

    const failed = results.filter((r) => r.failed).map((r) => r.category);
    setStatus_(ss, formatTime_(ss, now) + '｜本次新增 ' + newRows.length + ' 筆、已存在 ' + split.seenCount + ' 筆（沒動）' +
      (failed.length ? '｜' + failed.join('、') + ' 這次沒抓到，下次再試' : ''));
  } catch (err) {
    setStatus_(ss, '出錯了：' + err.message + '（可以再勾一次試試）');
    throw err;
  } finally {
    lock.releaseLock();
  }
}

function readSettings_(ss) {
  const sh = ss.getSheetByName(SHEETS.settings);
  let days = Number(sh.getRange(CELLS.days).getValue());
  if (!(days >= 1 && days <= 30)) days = DEFAULT_DAYS;
  const start = CATEGORY_HEADER_ROW + 1;
  const lastRow = sh.getLastRow();
  const categories = lastRow < start ? [] : sh.getRange(start, 1, lastRow - start + 1, 2).getValues()
    .map((r) => ({ name: String(r[0]).trim(), words: parseWords_(r[1]) }))
    .filter((c) => c.name && c.words.length);
  return { days: Math.round(days), categories: categories };
}

function readDataRows_(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  return sheet.getRange(2, 1, lastRow - 1, DATA_HEADERS.length).getValues();
}

function toRow_(item, now) {
  const published = isNaN(item.published.getTime()) ? '' : item.published;
  return [item.id, item.title, item.source, published, now, item.group, item.tags];
}

/** ⑤ 寫入並亮燈：新的放最上面、塗黃色；上一批的黃燈關掉 */
function writeResults_(sheet, newRows, recentWithGroups) {
  const lastRow = sheet.getLastRow();
  if (lastRow > 1) sheet.getRange(2, 1, lastRow - 1, DATA_HEADERS.length).setBackground(null); // 只改顏色，不改內容
  if (recentWithGroups) {
    // 舊資料被配成一組的，補上群組編號（只動加工欄，原始資料不動）
    sheet.getRange(2, COL.group + 1, recentWithGroups.length, 1).setValues(recentWithGroups.map((r) => [r.group]));
  }
  if (!newRows.length) return;
  sheet.insertRowsBefore(2, newRows.length);
  sheet.getRange(2, 1, newRows.length, DATA_HEADERS.length)
    .setValues(newRows)
    .setBackground(COLORS.fresh)
    .setFontWeight('normal')
    .setFontColor(COLORS.text);
  sheet.getRange(2, COL.published + 1, newRows.length, 2).setNumberFormat('yyyy/mm/dd hh:mm');
}

function writeBoard_(ss, summary) {
  const sh = ss.getSheetByName(SHEETS.board);
  sh.clear();
  sh.getRange('A1').setValue('關聯看板（每次抓完自動更新，這一頁不用改）')
    .setFontSize(14).setFontWeight('bold').setFontColor(COLORS.blue);

  const max = Math.max(1, ...summary.counts.map((c) => c.total));
  const catRows = [['分類', '累積則數', '本次新抓到', '比例']].concat(
    summary.counts.map((c) => ['【' + c.name + '】', c.total, c.fresh, '█'.repeat(Math.round((c.total / max) * 20))]));
  sh.getRange(3, 1, catRows.length, 4).setValues(catRows);
  sh.getRange(3, 1, 1, 4).setBackground(COLORS.blue).setFontColor('#FFFFFF').setFontWeight('bold');
  sh.getRange(4, 4, catRows.length - 1, 1).setFontColor(COLORS.blue);

  const top = 3 + catRows.length + 1;
  sh.getRange(top, 1).setValue('同一件事：被 2 家以上媒體報導（依則數排序，最多 30 組）').setFontWeight('bold');
  const groupRows = [['群組', '則數', '分類', '代表標題', '報導媒體']].concat(
    summary.topGroups.length
      ? summary.topGroups.map((g) => [g.group, g.count, g.tags, g.title, g.sources])
      : [['', '', '', '目前還沒有同一件事被多家媒體報導', '']]);
  sh.getRange(top + 1, 1, groupRows.length, 5).setValues(groupRows);
  sh.getRange(top + 1, 1, 1, 5).setBackground(COLORS.orange).setFontColor('#FFFFFF').setFontWeight('bold');
  [90, 80, 160, 440, 260].forEach((w, i) => sh.setColumnWidth(i + 1, w));
}

  ScriptApp.getProjectTriggers()
    .filter((t) => t.getHandlerFunction() === 'runDaily')
    .forEach((t) => ScriptApp.deleteTrigger(t));

}

/** 從一段 RSS 裡拿出某個標籤的內容 */
function pickTag_(block, tag) {
  const m = block.match(new RegExp('<' + tag + '(?:\\s[^>]*)?>([\\s\\S]*?)</' + tag + '>'));
  if (!m) return '';
  return decodeEntities_(m[1].replace(/^<!\[CDATA\[|\]\]>$/g, '')).trim();
}

/** 把 &amp; 這類跳脫字元還原成原本的字 */
function decodeEntities_(s) {
  const named = { amp: '&', lt: '<', gt: '>', quot: '"', apos: "'", nbsp: ' ' };
  return s.replace(/&(#x[0-9a-f]+|#\d+|amp|lt|gt|quot|apos|nbsp);/gi, (m, e) => {
    const k = e.toLowerCase();
    if (k[0] !== '#') return named[k];
    return String.fromCodePoint(k[1] === 'x' ? parseInt(k.slice(2), 16) : parseInt(k.slice(1), 10));
  });
}

function formatTime_(ss, date) {
  return Utilities.formatDate(date, ss.getSpreadsheetTimeZone(), 'M/d HH:mm');
}
